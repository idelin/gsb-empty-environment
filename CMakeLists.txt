cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(safelog safelog.cpp)
target_include_directories(safelog PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(safelog_test tests/safelog_test.cpp)
target_link_libraries(safelog_test PRIVATE safelog)
enable_testing()
add_test(NAME safelog_test COMMAND safelog_test)
