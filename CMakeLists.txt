cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
add_executable(test_mte tests/test_mte.cpp)
target_link_libraries(test_mte PRIVATE netsec)
find_package(Threads REQUIRED)
target_link_libraries(test_mte PRIVATE Threads::Threads)
add_test(NAME mte_policy COMMAND test_mte)
