cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
add_executable(test_slot_ownership tests/test_slot_ownership.cpp)
target_link_libraries(test_slot_ownership PRIVATE netsec)
add_test(NAME slot_ownership COMMAND test_slot_ownership)

add_executable(bus_monitor_example examples/bus_monitor_example.cpp)
target_link_libraries(bus_monitor_example PRIVATE netsec)
