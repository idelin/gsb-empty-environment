cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp source_trust.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
add_executable(test_source_trust tests/test_source_trust.cpp)
target_link_libraries(test_source_trust PRIVATE netsec)
add_test(NAME source_trust COMMAND test_source_trust)
