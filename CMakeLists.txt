cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(netsec PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(netsec PRIVATE /W4)
else()
    target_compile_options(netsec PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()

add_executable(free_access_test tests/free_access_test.cpp)
target_link_libraries(free_access_test PRIVATE netsec)
if(MSVC)
    target_compile_options(free_access_test PRIVATE /W4)
else()
    target_compile_options(free_access_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME free_access_test COMMAND free_access_test)
