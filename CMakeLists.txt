cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Threads REQUIRED)
target_link_libraries(netsec PUBLIC Threads::Threads)
enable_testing()
add_executable(session_tests tests/session_tests.cpp)
target_link_libraries(session_tests PRIVATE netsec)
add_test(NAME session_tests COMMAND session_tests)
