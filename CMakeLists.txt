cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec OBJECT netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()

add_executable(test_guard test_guard.cpp)
target_link_libraries(test_guard PRIVATE netsec)
target_compile_options(netsec PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(test_guard PRIVATE -Wall -Wextra -Wpedantic)

add_test(NAME pointer_guard COMMAND test_guard)
