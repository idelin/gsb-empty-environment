cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
add_executable(trace_permission_tests tests/trace_permission_tests.cpp)
target_link_libraries(trace_permission_tests PRIVATE netsec)
add_test(NAME trace_permission_tests COMMAND trace_permission_tests)
