cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_sources(netsec PRIVATE ndef.cpp)
target_compile_options(netsec PRIVATE -Wall -Wextra -Wpedantic)

enable_testing()
add_executable(ndef_test ndef_test.cpp ndef.cpp)
target_compile_options(ndef_test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME ndef_test COMMAND ndef_test)
