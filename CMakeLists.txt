cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp anti_rollback.cpp sha256.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
add_executable(anti_rollback_tests tests/anti_rollback_tests.cpp)
target_link_libraries(anti_rollback_tests PRIVATE netsec)
add_test(NAME anti_rollback COMMAND anti_rollback_tests)
