cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
add_executable(bitmap_parser_test tests/bitmap_parser_test.cpp)
target_link_libraries(bitmap_parser_test PRIVATE netsec)
add_test(NAME bitmap_parser_test COMMAND bitmap_parser_test)
