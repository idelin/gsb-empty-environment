cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
add_executable(pairing_policy_test tests/pairing_policy_test.cpp)
target_link_libraries(pairing_policy_test PRIVATE netsec)
add_test(NAME pairing_policy_test COMMAND pairing_policy_test)
