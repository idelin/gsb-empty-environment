cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(netsec PUBLIC cxx_std_20)
add_library(netsec::netsec ALIAS netsec)

if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    set(NETSEC_BUILD_TESTS_DEFAULT ON)
else()
    set(NETSEC_BUILD_TESTS_DEFAULT OFF)
endif()
option(NETSEC_BUILD_TESTS "Build the fail-stop voter tests" ${NETSEC_BUILD_TESTS_DEFAULT})
if(NETSEC_BUILD_TESTS)
    enable_testing()
    add_executable(netsec_vote_tests tests/voting_tests.cpp)
    target_link_libraries(netsec_vote_tests PRIVATE netsec)
    add_test(NAME netsec_vote_tests COMMAND netsec_vote_tests)
endif()
