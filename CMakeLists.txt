cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()

option(NETSEC_BUILD_TESTS "Build netsec validation tests" ON)
if(NETSEC_BUILD_TESTS)
    add_executable(netsec_test tests/netsec_test.cpp)
    target_link_libraries(netsec_test PRIVATE netsec)
    add_test(NAME netsec_test COMMAND netsec_test)

    add_executable(netsec_fuzz tests/netsec_fuzz.cpp)
    target_link_libraries(netsec_fuzz PRIVATE netsec)
    add_test(NAME netsec_fuzz COMMAND netsec_fuzz)
endif()
