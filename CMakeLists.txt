cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec netsec.cpp)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
find_package(Threads REQUIRED)
add_executable(test_partition test_partition.cpp)
target_link_libraries(test_partition PRIVATE netsec Threads::Threads)
add_test(NAME partition COMMAND test_partition)
