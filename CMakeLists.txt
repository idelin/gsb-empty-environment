cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(APPLE)
    if(CMAKE_AR STREQUAL "/usr/bin/ar" AND EXISTS "/Library/Developer/CommandLineTools/usr/bin/ar")
        set(CMAKE_AR "/Library/Developer/CommandLineTools/usr/bin/ar" CACHE FILEPATH "" FORCE)
    endif()
    if(CMAKE_RANLIB STREQUAL "/usr/bin/ranlib" AND EXISTS "/Library/Developer/CommandLineTools/usr/bin/ranlib")
        set(CMAKE_RANLIB "/Library/Developer/CommandLineTools/usr/bin/ranlib" CACHE FILEPATH "" FORCE)
    endif()
endif()

add_library(netsec netsec.cpp)
add_library(netsec::netsec ALIAS netsec)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(netsec PUBLIC cxx_std_20)

if(MSVC)
    target_compile_options(netsec PRIVATE /W4)
else()
    target_compile_options(netsec PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(netsec_tests tests/ats_isolation_tests.cpp)
    target_link_libraries(netsec_tests PRIVATE netsec Threads::Threads)
    if(MSVC)
        target_compile_options(netsec_tests PRIVATE /W4)
    else()
        target_compile_options(netsec_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME ats_isolation COMMAND netsec_tests)
endif()
