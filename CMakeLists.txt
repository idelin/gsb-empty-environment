cmake_minimum_required(VERSION 3.16)
project(netsec LANGUAGES CXX)
if(APPLE AND EXISTS "/Library/Developer/CommandLineTools/usr/bin/ar")
  set(CMAKE_AR "/Library/Developer/CommandLineTools/usr/bin/ar" CACHE FILEPATH "" FORCE)
  if(EXISTS "/Library/Developer/CommandLineTools/usr/bin/ranlib")
    set(CMAKE_RANLIB "/Library/Developer/CommandLineTools/usr/bin/ranlib" CACHE FILEPATH "" FORCE)
  endif()
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netsec STATIC netsec.cpp)
target_compile_features(netsec PUBLIC cxx_std_20)
target_include_directories(netsec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()

add_executable(netsec_component_tests tests/netsec_tests.cpp)
target_link_libraries(netsec_component_tests PRIVATE netsec)
add_test(NAME netsec_component_tests COMMAND netsec_component_tests)
